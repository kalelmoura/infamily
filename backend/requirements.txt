# Dependencies are added in their roadmap phases.
#
# uvicorn is intentionally the plain build (not uvicorn[standard]) to avoid
# compiled extras (uvloop/httptools/watchfiles) that may lack wheels on very
# new Python versions. --reload still works via a stat-based fallback reloader.

fastapi
uvicorn
pydantic-settings
PyJWT[crypto]
# The [asyncio] extra pulls in greenlet, which SQLAlchemy's async layer needs
# to bridge its (synchronous) internals to async I/O. Plain `sqlalchemy` does
# not always install it — on Python 3.14 it did not, and every session close
# raised "the greenlet library is required". Always use the extra for async.
sqlalchemy[asyncio]
asyncpg
alembic
# Calendar-aware date math for fiado installments. Adding "one month" to a due
# date is not adding 30 days — relativedelta knows that 31 Jan + 1 month is the
# last day of February, which naive arithmetic gets wrong every single time.
python-dateutil

# --- Phase 8: product photos ---
# FastAPI parses multipart/form-data (what a file upload is) through this
# package. It is not a FastAPI dependency by default, and without it an
# endpoint declaring `UploadFile` raises at import time, not at request time.
python-multipart
# Decoding an upload is the only real proof it is an image — a Content-Type
# header and a file extension are both client-supplied and both lie. Pillow
# also fixes the phone-rotation flag and strips EXIF (GPS included) on the
# re-encode, which matters because the photo bucket is public-read.
Pillow
# Async HTTP client for the Supabase Storage REST API. Deliberately not the
# `supabase` package: it drags in gotrue, postgrest, realtime and storage3 to
# do what is, here, exactly two HTTP calls.
httpx
