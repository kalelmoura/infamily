# The [asyncio] extra pulls in greenlet, which SQLAlchemy's async layer needs
# to bridge its (synchronous) internals to async I/O. Plain `sqlalchemy` does
# not always install it — on Python 3.14 it did not, and every session close
# raised "the greenlet library is required". Always use the extra for async.
sqlalchemy[asyncio]
asyncpg
alembic
# Calendar-aware date math for fiado installments. Adding "one month" to a due
# date is not adding 30 days — relativedelta knows that 31 Jan + 1 month is the
# last day of February, which naive arithmetic gets wrong every single time.
python-dateutil
